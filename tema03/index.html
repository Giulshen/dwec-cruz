Ejercicio 1

Variables y typeof

function ejercicio1() {
  // Muestro en la consola el nombre del ejercicio
  console.log("--- Ejercicio 1 · Variables y typeof ---");

  // Uso const porque el valor de estas variables no cambia
  const edad = 20;
  // Muestro el valor de edad y su tipo de dato
  console.log("edad =", edad, "→", typeof edad);

  const curso = "2º DAW";
  // Las palabras y los textos son de tipo string
  console.log("curso =", curso, "→", typeof curso);

  const estudiaDaw = true;
  // true y false son valores de tipo boolean
  console.log("estudiaDaw =", estudiaDaw, "→", typeof estudiaDaw);

  const premio = null;
  // null significa que no hay ningún valor asignado intencionadamente
  console.log("premio =", premio, "→", typeof premio);

  const numeroGrande = 10n;
  // La n al final indica que es un número de tipo bigint
  console.log("numeroGrande =", numeroGrande, "→", typeof numeroGrande);

  // Uso let porque después voy a darle un valor a esta variable
  let horasEstudiadas;
  // Al no tener valor, su tipo es undefined
  console.log("horasEstudiadas =", horasEstudiadas, "→", typeof horasEstudiadas);

  // Le asigno un valor y compruebo cómo cambia su tipo
  horasEstudiadas = 6;
  console.log("horasEstudiadas =", horasEstudiadas, "→", typeof horasEstudiadas);
}

Ejercicio 2

Conversiones explícitas

function ejercicio2() {
  // Muestro en la consola el nombre del ejercicio
  console.log("--- Ejercicio 2 · Conversiones explícitas ---");

  // Con String convierto el número 123 en un texto
  const a = String(123);
  console.log("String(123) →", a, typeof a);

  // Con Number convierto el texto "123" en un número
  const b = Number("123");
  console.log('Number("123") →', b, typeof b);

  // Como "12abc" no es un número válido, el resultado es NaN
  const c = Number("12abc");
  console.log('Number("12abc") →', c, typeof c);

  // Una cadena vacía se convierte en el número 0
  const d = Number("");
  console.log('Number("") →', d, typeof d);

  // El valor true se convierte en el número 1
  const e = Number(true);
  console.log("Number(true) →", e, typeof e);

  // El número 0 se convierte en false
  const f = Boolean(0);
  console.log("Boolean(0) →", f, typeof f);

  // Un texto que no está vacío se convierte en true
  const g = Boolean("texto");
  console.log('Boolean("texto") →', g, typeof g);

  // Una cadena vacía se convierte en false
  const h = Boolean("");
  console.log('Boolean("") →', h, typeof h);

  // Con String convierto null en el texto "null"
  const i = String(null);
  console.log("String(null) →", i, typeof i);

  // undefined no se puede convertir en un número válido, por eso sale NaN
  const j = Number(undefined);
  console.log("Number(undefined) →", j, typeof j);
}

Ejercicio 3

Coerción y comparaciones

function ejercicio3() {
  // Muestro en la consola el nombre del ejercicio
  console.log("--- Ejercicio 3 · Coerción y comparaciones ---");

  // Pruebo qué pasa al mezclar números, textos y valores booleanos
  console.log('"5" - 2 →', "5" - 2);   // El texto se convierte en número
  console.log('"5" + 2 →', "5" + 2);   // El número se junta con el texto
  console.log("true + 1 →", true + 1);   // true equivale a 1
  console.log('"3" + 4 + 5 →', "3" + 4 + 5);   // Se van juntando como texto
  console.log('4 + 5 + "3" →', 4 + 5 + "3");   // Primero suma y después junta
  console.log('"hola" - 1 →', "hola" - 1);   // No puede restar y devuelve NaN

  // Con == se comparan los valores permitiendo conversiones de tipo
  console.log('5 == "5" →', 5 == "5");     // Devuelve true

  // Con === se comparan el valor y el tipo de dato
  console.log('5 === "5" →', 5 === "5");   // Devuelve false

  // El 0 y false se consideran iguales con ==
  console.log("0 == false →", 0 == false);

  // Con === son distintos porque sus tipos no coinciden
  console.log("0 === false →", 0 === false);

  // null y undefined son iguales al usar ==
  console.log("null == undefined →", null == undefined);

  // Con === son diferentes porque son tipos distintos
  console.log("null === undefined →", null === undefined);
}

Cuándo manda la cadena y cuándo el número: con + manda la cadena: si uno de los lados es texto, JavaScript junta en vez de sumar (por eso "5" + 2 da "52", y 4 + 5 + "3" da "93", porque primero suma 4 + 5 y después junta). Con -, * o / manda el número: intenta convertir el texto en número y, si no puede, sale NaN. Por eso comparo siempre con ===, que no convierte los tipos y evita sorpresas como 0 == false.

Ejercicio 4

Tu ficha con plantillas de cadena

function ejercicio4() {
  // Muestro en la consola el nombre del ejercicio
  console.log("--- Ejercicio 4 · Tu ficha con plantillas de cadena ---");

  // Guardo mis datos en constantes porque no necesito cambiarlos
  const nombre = "Giulshen Hueso Cruz";
  const ciclo = "Desarrollo de Aplicaciones Web";
  const curso = "2º curso";
  const aficion = "la fotografía";

  // Uso let porque las horas estudiadas van a cambiar
  let horasEstudiadas = 5;

  // Sumo tres horas a las cinco que tenía inicialmente
  horasEstudiadas += 3;

  // Creo una frase usando backticks y ${} para incluir las variables
  const ficha = `Soy ${nombre}, estudio ${ciclo} en ${curso} y mi afición es ${aficion}. Esta semana he estudiado ${horasEstudiadas} horas.`;

  // Muestro la ficha en una ventana emergente y también en la consola
  alert(ficha);
  console.log(ficha);

  // Creo la misma frase utilizando el operador + para unir los textos
  const fichaConMas = "Soy " + nombre + ", estudio " + ciclo + " en " + curso + " y mi afición es " + aficion + ". Esta semana he estudiado " + horasEstudiadas + " horas.";
  console.log(fichaConMas);

  // Comparo las dos frases para comprobar si son exactamente iguales
  console.log("¿Son iguales las dos fichas? →", ficha === fichaConMas);
}

Uncaught TypeError: Assignment to constant variable.

Qué significa: he intentado darle otro valor a una variable declarada con const, y JavaScript no lo permite, porque una const se queda con el valor con el que nace. Si el valor tiene que cambiar, hay que declararla con let.
